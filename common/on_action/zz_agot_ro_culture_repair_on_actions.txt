# Runs the culture repair so characters that end up with a deleted culture get
# fixed before they flood error.log or crash the game. Some ruler's quarterly
# pulse fires almost every day, so this runs within a day or two of loading a
# save, then at most once a month.
quarterly_playable_pulse = {
	on_actions = { agot_ro_culture_repair_pulse }
}

agot_ro_culture_repair_pulse = {
	trigger = {
		NOT = { has_global_variable = agot_ro_culture_repair_recent }
	}
	effect = {
		set_global_variable = {
			name = agot_ro_culture_repair_recent
			value = yes
			days = 30
		}
		agot_ro_repair_broken_cultures_effect = yes
	}
}
